function switchTab(tabId) {

document.querySelectorAll('.section').forEach(s => s.classList.remove('active'));

document.querySelectorAll('nav button').forEach(b => b.classList.remove('active'));

document.getElementById(tabId).classList.add('active');

document.getElementById('btn-' + tabId).classList.add('active');

}


function loadData() {

// Load Service Logs

const services = JSON.parse(localStorage.getItem('wrenchlog_services') || '[]');

const serviceList = document.getElementById('service-list');

serviceList.innerHTML = services.map(s => `

<div class="item-card">

<strong>${s.title}</strong> (${s.vehicle})

<p style="margin: 5px 0 0 0; color: var(--text-secondary);">${s.notes}</p>

</div>

`).join('') || '<p style="color: var(--text-secondary);">No service records yet.</p>';


// Load Vehicles

const vehicles = JSON.parse(localStorage.getItem('wrenchlog_vehicles') || '[]');

const vehicleList = document.getElementById('vehicle-list');

vehicleList.innerHTML = vehicles.map(v => `

<div class="item-card">

<strong>${v.name}</strong>

<p style="margin: 5px 0 0 0; color: var(--text-secondary);">${v.engine}</p>

</div>

`).join('') || '<p style="color: var(--text-secondary);">No vehicles added yet.</p>';

}


function addServiceRecord() {

const title = document.getElementById('service-title').value;

const vehicle = document.getElementById('service-vehicle').value;

const notes = document.getElementById('service-notes').value;

if(!title) return alert('Please enter a service title.');


const services = JSON.parse(localStorage.getItem('wrenchlog_services') || '[]');

services.unshift({ title, vehicle, notes, date: new Date().toLocaleDateString() });

localStorage.setItem('wrenchlog_services', JSON.stringify(services));

document.getElementById('service-title').value = '';

document.getElementById('service-vehicle').value = '';

document.getElementById('service-notes').value = '';

loadData();

}


function addVehicle() {

const name = document.getElementById('veh-year-make-model').value;

const engine = document.getElementById('veh-engine').value;

if(!name) return alert('Please enter the vehicle details.');


const vehicles = JSON.parse(localStorage.getItem('wrenchlog_vehicles') || '[]');

vehicles.unshift({ name, engine });

localStorage.setItem('wrenchlog_vehicles', JSON.stringify(vehicles));

document.getElementById('veh-year-make-model').value = '';

document.getElementById('veh-engine').value = '';

loadData();

}
